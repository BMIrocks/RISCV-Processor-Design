# ========================================================================
# 4-HAZARD DEMONSTRATION INSTRUCTIONS
# ========================================================================
# This program demonstrates all 4 types of pipeline hazards

# === SETUP: Initialize test values ===
addi x1, x0, 10      # x1 = 10
addi x2, x0, 20      # x2 = 20
addi x3, x0, 5       # x3 = 5
add x0, x0, x0       # NOP

# ========================================================================
# HAZARD 1: EX DATA HAZARD (Immediate dependency - EX to EX forwarding)
# ========================================================================
# The producer instruction is in EX stage when consumer is in ID
# Forwarding from EX/MEM register resolves this (forward_A or forward_B = 10)

add x4, x1, x2       # Cycle N:   x4 = 10+20 = 30 (producer)
add x5, x4, x3       # Cycle N+1: x5 = x4+5     (consumer - NEEDS x4 from EX)
                     # >>> EX HAZARD: forward_A = 10 to get x4 from EX/MEM

# ========================================================================
# HAZARD 2: MEM DATA HAZARD (2-instruction gap - MEM to EX forwarding)
# ========================================================================
# The producer instruction is in MEM stage when consumer is in ID
# Forwarding from MEM/WB register resolves this (forward_A or forward_B = 01)

add x6, x1, x2       # Cycle M:   x6 = 10+20 = 30 (producer)
add x0, x0, x0       # Cycle M+1: NOP (creates gap)
add x7, x6, x3       # Cycle M+2: x7 = x6+5     (consumer - NEEDS x6 from MEM)
                     # >>> MEM HAZARD: forward_A = 01 to get x6 from MEM/WB

# ========================================================================
# HAZARD 3: LOAD-USE HAZARD (Load followed by immediate use - STALL)
# ========================================================================
# Load instruction cannot provide data in time for next instruction
# Must stall 1 cycle and insert bubble

sd x1, 0(x0)         # Store x1=10 to memory[0]
ld x8, 0(x0)         # Cycle L:   x8 = mem[0] = 10 (load - data ready in MEM)
add x9, x8, x2       # Cycle L+1: x9 = x8+20 (consumer - NEEDS x8 but not ready!)
                     # >>> LOAD-USE HAZARD: stall_IF=1, stall_ID=1, flush_ID_EX=1
                     # Pipeline inserts 1-cycle bubble, then forwards from MEM

# Verify load-use with second example
sd x2, 8(x0)         # Store x2=20 to memory[8]
ld x10, 8(x0)        # x10 = mem[8] = 20
sub x11, x10, x3     # x11 = x10-5 (another load-use hazard)

# ========================================================================
# HAZARD 4: CONTROL HAZARD (Branch - flush pipeline)
# ========================================================================
# Branch decision not known until EX stage
# If branch taken, must flush instructions in IF and ID stages

addi x12, x0, 100    # x12 = 100 (will be flushed if branch taken)
beq x1, x1, 12       # Branch if x1==x1 (always true - BRANCH TAKEN!)
                     # >>> CONTROL HAZARD: flush_IF_ID=1
addi x13, x0, 99     # x13 = 99  (FLUSHED - should not execute)
addi x14, x0, 88     # x14 = 88  (FLUSHED - should not execute)
# Branch target (PC+12):
addi x15, x0, 77     # x15 = 77  (branch target - executes)

# Another branch - NOT taken
addi x16, x0, 50     # x16 = 50
bne x1, x1, 8        # Branch if x1!=x1 (false - NOT TAKEN)
addi x17, x0, 60     # x17 = 60 (executes - no flush)

# ========================================================================
# COMBINED HAZARDS: Multiple hazards in sequence
# ========================================================================

ld x18, 0(x0)        # Load x18 = mem[0] = 10
add x19, x18, x1     # Load-use hazard + EX forwarding after stall
add x20, x19, x2     # EX hazard - forward from previous add

# ========================================================================
# END MARKER
# ========================================================================
add x0, x0, x0       # NOP - halt
